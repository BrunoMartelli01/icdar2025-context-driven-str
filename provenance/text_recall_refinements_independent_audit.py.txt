def matches(predictions, truth, require_text):
    assigned = {}
    edges = [[i for i, word in enumerate(truth) if box_iou(p['bbox'], word['bbox']) >= .5
              and (not require_text or p['text'].casefold() == word['text'].casefold())]
             for p in predictions]
    def visit(pred_index, seen):
        for word_index in edges[pred_index]:
            if word_index in seen:
                continue
            seen.add(word_index)
            if word_index not in assigned or visit(assigned[word_index], seen):
                assigned[word_index] = pred_index
                return True
        return False
    for i in range(len(predictions)):
        visit(i, set())
    return len(assigned)


checked_files = checked_crops = checked_rows = 0
images = {}
populations = []
for branch, methods in g['branches'].items():
    for method, part in methods.items():
        recognition = r['branches'][branch][method]
        assert len(part['per_image']) == len(recognition['per_image']) == 125
        populations.append([x['sample_id'] for x in part['per_image']])
        counts = Counter()
        for row, ocr in zip(part['per_image'], recognition['per_image']):
            assert row['sample_id'] == ocr['sample_id']
            assert row['ground_truth'] == ocr['ground_truth']
            sid = row['sample_id']
            if sid not in images:
                assert sha256_file(Path(row['image_path'])) == row['image_sha256']
                assert sha256_file(Path(row['annotation_path'])) == row['annotation_sha256']
                with Image.open(row['image_path']) as im:
                    images[sid] = np.asarray(im.convert('RGB')).copy()
            for filename, expected in row['exports'].items():
                assert sha256_file(ROOT/filename) == expected, filename
                checked_files += 1
            assert len(row['proposals']) == len(ocr['predictions_all'])
            ignored = set(ocr['ignored_prediction_ids'])
            care = []
            for p, predicted in zip(row['proposals'], ocr['predictions_all']):
                assert p['bbox'] == predicted['bbox']
                x1, y1, x2, y2 = p['bbox']
                assert crop_array_hash(images[sid][y1:y2, x1:x2]) == p['crop_rgb_sha256']
                record = json.loads((ROOT/'ocr'/(p['crop_rgb_sha256']+'.json')).read_text())
                assert predicted['text'] == record['raw_text']
                if p['instance_id'] not in ignored:
                    care.append(predicted)
                checked_crops += 1
            current = {'gt_words': len(row['ground_truth']), 'proposals': len(care),
                'geometry_iou_ge_50': matches(care, row['ground_truth'], False),
                'exact_casefold_iou_ge_50': matches(care, row['ground_truth'], True)}
            assert all(current[k] == ocr['counts'][k] for k in current)
            counts.update(current)
            checked_rows += 1
        a = recognition['aggregate']
        assert counts['gt_words'] == 983
        assert all(counts[k] == a[k] for k in counts)
        tp, proposals, total = counts['exact_casefold_iou_ge_50'], counts['proposals'], counts['gt_words']
        assert a['exact_casefold_precision'] == (tp/proposals if proposals else 0.)
        assert a['exact_casefold_recall_all_gt'] == tp/total
        assert a['exact_casefold_f1'] == 2*tp/(proposals+total)
        print(branch, method, f'{tp}/{total}', f'Hmean={a["exact_casefold_f1"]:.6f}', flush=True)
assert all(p == populations[0] for p in populations)
for key, expected in r['ocr_records_sha256'].items():

assert g.get('original_native_geometry_replays', 0) == (500 if len(populations) == 32 else 0)
